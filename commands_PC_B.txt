docker build -t ev_engine:local ./ev_cp_engine

# Arrancar Engine
docker run --rm -p 5001:5001 --name engine `
  -e ENGINE_PORT=5001 -e CP_ID=CP_001 `
  -e KAFKA_SERVER="<CENTRAL_IP>:9092" `
  ev_engine:local

# Arrancar Monitor
docker run --rm --name monitor `
  -e CP_ID=CP_001 `
  -e CENTRAL_IP=<CENTRAL_IP> -e CENTRAL_PORT=5000 `
  -e ENGINE_IP=172.27.237.31 -e ENGINE_PORT=5001 `
  ev_monitor:local

# Arrancar Driver
docker run --rm --name driver `
  -e KAFKA_BROKER="<CENTRAL_IP>:9092" `
  -e DRIVER_ID=DRIVER_456 -e CP_ID=CP_001 -e MAT=ABC-1234 -e KW=25.0 -e LISTEN=true `
  ev_driver:local
